    ```typescript
    // src/lib/habitat-3d/mood-decay.ts
    import * as THREE from "three";
    import type { TigerMood } from "@/lib/habitat-engine";
    import type { ClayWorld } from "./clay-world";

    const MOOD_TARGETS: Record<TigerMood, { speedMul: number; headDroop: number; sparkleOn: boolean }> = {
      excited: { speedMul: 2.2, headDroop: 0,    sparkleOn: true  },  // D-07: bounce + sparkles
      happy:   { speedMul: 1.5, headDroop: 0,    sparkleOn: false },  // designer default
      neutral: { speedMul: 0.9, headDroop: 0.05, sparkleOn: false },
      sad:     { speedMul: 0.5, headDroop: 0.15, sparkleOn: false },
    };

    // D-06: bounce duration for "happier" transitions
    const BOUNCE_MS = 300;
    // D-06: crossfade duration for "sadder" transitions (lerp toward target)
    const CROSSFADE_RATE = 1 / 0.5; // 0.5 second crossfade

    const MOOD_ORDER: TigerMood[] = ["sad", "neutral", "happy", "excited"];
    function isHappierShift(prev: TigerMood, next: TigerMood): boolean {
      return MOOD_ORDER.indexOf(next) > MOOD_ORDER.indexOf(prev);
    }

    let prevMood: TigerMood | null = null;
    export function applyMood(world, mood, state, now) {
      const target = MOOD_TARGETS[mood];
      if (prevMood !== null && prevMood !== mood) {
        if (isHappierShift(prevMood, mood)) {
          // D-06: snap + bounce
          state.speedMul    = target.speedMul;
          state.headDroop   = target.headDroop;
          state.sparkleOn   = target.sparkleOn;
          state.bounceUntil = now + BOUNCE_MS;
        } else {
          // D-06: crossfade (lerp speedMul + headDroop over CROSSFADE_RATE seconds)
          // (driven by world's dt — store target and lerp in subsequent ticks)
          state.targetSpeedMul  = target.speedMul;
          state.targetHeadDroop = target.headDroop;
          state.sparkleOn       = target.sparkleOn;
          // actual lerp is performed inside updateWorld using dt
        }
      } else if (prevMood === null) {
        Object.assign(state, target);
        state.bounceUntil = 0;
      }
      prevMood = mood;

      // Apply head droop (idempotent — clay-animation.ts reads state.headDroop)
      if (world.characters.lion?.headG) {
        world.characters.lion.headG.userData.moodDroop = state.headDroop;
      }

      // Bounce on root.position.y for happy shift
      if (now < state.bounceUntil) {
        const phase = (state.bounceUntil - now) / BOUNCE_MS;
        const bounce = Math.sin((1 - phase) * Math.PI * 6) * 0.15 * phase;
        if (world.characters.lion?.root) {
          world.characters.lion.root.userData.moodBounce = bounce;
        }
      }
    }

    // Decay — RESEARCH section C.3
    const COLOR_DECAYED = new THREE.Color("#6a7560");  // dull olive
    const COLOR_SKY_DECAYED = new THREE.Color("#a8b5b8");
    export function applyDecay(world, quality) {
      const q = Math.max(0, Math.min(1, quality));
      const tier2 = q < 0.4;

      // Material color lerp on flowers + grass + ground accent materials
      world.featureGroups.flowers?.traverse((o) => {
        const m = (o as any).material as THREE.MeshStandardMaterial | undefined;
        if (m?.color) m.color.copy(m.userData.baseColor ?? m.color).lerp(COLOR_DECAYED, 1 - q);
      });

      // Fog tightening (Plan 02 must store baseline on world.fog.userData)
      if (world.fog) {
        world.fog.near = 20 + (1 - q) * 16;
        world.fog.far  = 60 + (1 - q) * 20;
      }

      // Tier-2 visibility cull
      if (world.featureGroups.butterflies) world.featureGroups.butterflies.visible = !tier2;
      if (world.featureGroups.flowers)     world.featureGroups.flowers.visible     = !tier2;

      // Sky tint (only when tier2)
      if (world.sky?.uniforms?.color1) {
        if (tier2) world.sky.uniforms.color1.value.copy(COLOR_SKY_DECAYED);
        else       world.sky.uniforms.color1.value.copy(world.sky.userData.baseColor1);
      }
    }
    ```

    NOTE: Plan 02 must have stored `baseColor` on each affected
    material's `userData` (and `baseColor1` on the sky shader uniform's
    `userData`) so this lerp is reversible. If Plan 02 did not, this
    plan must add that bookkeeping inside `buildClayWorld` (small edit
    to clay-world.ts: when constructing flower/sky materials, save
    `material.userData.baseColor = material.color.clone()`).

    Per RESEARCH Pitfall 5, do NOT use post-processing / EffectComposer.
    All decay is per-material/per-fog mutation. This is the GPU-cheap path.
