    ```typescript
    "use client";
    import { useEffect, useRef, useState } from "react";
    import { buildSceneHost, attachOrbit } from "@/lib/habitat-3d/scene-host";
    import { buildClayWorld } from "@/lib/habitat-3d/clay-world";
    import { featuresForLevel, LEVEL_CONFIG } from "@/lib/habitat-3d/feature-flags";
    import { updateWorld } from "@/lib/habitat-3d/clay-animation";
    import type { HabitatState } from "@/lib/habitat-engine";

    function usePrefersReducedMotion(): boolean { /* RESEARCH Example 2 */ }

    export default function HabitatCanvas({
      habitatState,
      celebratingLevel = null,
    }: HabitatCanvasProps) {
      const canvasRef    = useRef<HTMLCanvasElement>(null);
      const wrapperRef   = useRef<HTMLDivElement>(null);
      const stateRef     = useRef(habitatState);      // updated on every render
      stateRef.current   = habitatState;
      const reducedMotion = usePrefersReducedMotion();

      // structKey: only rebuild scene when level changes (per designer's pattern)
      const structKey = habitatState.level;

      useEffect(() => {
        if (!canvasRef.current || !wrapperRef.current) return;
        const disposedRef = { current: false };
        const ctx     = buildSceneHost(canvasRef.current);
        const features = featuresForLevel(habitatState.level);
        const cfg     = LEVEL_CONFIG[habitatState.level] ?? LEVEL_CONFIG[1];
        const world   = buildClayWorld(ctx, features, { sky: cfg.sky });
        const orbit   = attachOrbit(canvasRef.current, ctx.camera, {
          reducedMotion,
        });

        // R5: keyboard handler
        const wrapper = wrapperRef.current;
        wrapper.tabIndex = 0;
        const autoSpeed = 0.12;
        const keyStep   = autoSpeed * 0.016;
        const onKey = (e: KeyboardEvent) => {
          if (e.key === "ArrowLeft")  { orbit.nudgeTheta(-keyStep); e.preventDefault(); }
          if (e.key === "ArrowRight") { orbit.nudgeTheta( keyStep); e.preventDefault(); }
        };
        wrapper.addEventListener("keydown", onKey);

        // RAF loop with disposedRef guard (RESEARCH Pitfall 2)
        let raf = 0;
        let last = performance.now();
        const tick = (now: number) => {
          if (disposedRef.current) return;
          const dt = Math.min(0.05, (now - last) / 1000);
          last = now;
          orbit.tick(dt);
          updateWorld(ctx, world, {
            dt,
            t: now / 1000,
            reducedMotion,
            state: stateRef.current,        // Plan 04 binds via this
          });
          ctx.renderer.render(ctx.scene, ctx.camera);
          raf = requestAnimationFrame(tick);
        };
        raf = requestAnimationFrame(tick);

        // signal "ready" for screenshot tooling (R7, Plan 04) + e2e
        canvasRef.current.setAttribute("data-ready", "true");

        // WebGL context loss handling (RESEARCH section E.2, Pitfall 1)
        const onLost = (e: Event) => {
          e.preventDefault();
          cancelAnimationFrame(raf);
          canvasRef.current?.setAttribute("data-ready", "false");
        };
        const onRestored = () => {
          last = performance.now();
          raf = requestAnimationFrame(tick);
          canvasRef.current?.setAttribute("data-ready", "true");
        };
        canvasRef.current.addEventListener("webglcontextlost", onLost);
        canvasRef.current.addEventListener("webglcontextrestored", onRestored);

        return () => {
          disposedRef.current = true;
          cancelAnimationFrame(raf);
          wrapper.removeEventListener("keydown", onKey);
          canvasRef.current?.removeEventListener("webglcontextlost", onLost);
          canvasRef.current?.removeEventListener("webglcontextrestored", onRestored);
          orbit.dispose();
          world.dispose?.();
          // traverse + dispose geometries + materials per designer pattern
          ctx.scene.traverse((o) => {
            const m = (o as THREE.Mesh).material;
            const g = (o as THREE.Mesh).geometry;
            if (g) g.dispose();
            if (Array.isArray(m)) m.forEach((mm) => mm.dispose());
            else if (m) m.dispose();
          });
          ctx.renderer.dispose();
        };
      }, [structKey, reducedMotion]);   // ONLY level + reducedMotion trigger rebuild

      return (
        <div ref={wrapperRef} className="w-full h-full focus:outline-none focus:ring-2 focus:ring-primary rounded-lg" role="img" aria-label="Tiger habitat 3D scene">
          <canvas ref={canvasRef} data-testid="habitat-3d-canvas" />
        </div>
      );
    }
    ```

    - Default export (matches what `next/dynamic` calls expect)
    - No `window.*` access at module top level — all inside `useEffect`
    - Strict-Mode-safe cleanup (disposedRef + full traversal disposal)

    DO NOT bind mood or quality to scene materials in this plan — pass

    is Plan 04. For now, `updateWorld` may simply read
    `state.level` for sleep-cycle gating and ignore `mood`/`quality`.
