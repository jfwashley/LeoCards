```typescript
// src/lib/habitat-3d/types.ts (NEW)
import type * as THREE from "three";

export interface SceneContext {
  renderer: THREE.WebGLRenderer;
  scene:    THREE.Scene;
  camera:   THREE.PerspectiveCamera;
  canvas:   HTMLCanvasElement;
  width:    number;
  height:   number;
  isMobile: boolean;
  /** Designer's quality scalar — Q = isMobile ? 0.55 : 1 */
  Q:        number;
}

export interface OrbitOptions {
  lookY?:         number;   // default 1 (per habitats-shared.jsx:31)
  idleDelay?:     number;   // default 1200ms
  autoSpeed?:     number;   // default 0.12 rad/s
  reducedMotion?: boolean;  // NEW for R6
}

export interface OrbitHandle {
  tick:         (dt: number) => void;
  dispose:      () => void;
  /** NEW for R5 — keyboard handler calls this; one frame's worth of autoSpeed. */
  nudgeTheta:   (delta: number) => void;
  /** Read-only for tests + Playwright assertions. */
  getTheta:     () => number;
}

export interface WorldOpts {
  sky?: "default" | "golden-hour";
}
```
