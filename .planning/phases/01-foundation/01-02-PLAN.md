    1. Create `src/lib/auth.ts` — Better Auth server instance:
       ```typescript
       import { betterAuth } from "better-auth";
       import { drizzleAdapter } from "better-auth/adapters/drizzle";
       import { nextCookies } from "better-auth/next-js";
       import { db } from "@/db";
       import * as schema from "@/db/schema";

       export const auth = betterAuth({
         database: drizzleAdapter(db, {
           provider: "pg",
           schema,  // MUST include full schema with Better Auth tables
         }),
         emailAndPassword: {
           enabled: true,
           sendResetPassword: async ({ user, url }) => {
             // Import Resend inside the function to avoid top-level side effects
             const { Resend } = await import("resend");
             const resend = new Resend(process.env.RESEND_API_KEY);
             // void to prevent timing attacks
             void resend.emails.send({
               from: "TioCards <noreply@tiocards.com>",
               to: user.email,
               subject: "Reset your TioCards password",
               text: `Reset your password: ${url}`,
             });
           },
           revokeSessionsOnPasswordReset: true,
         },
         plugins: [nextCookies()],  // MUST be last plugin in array
       });
       ```

       **Critical:** The `schema` import MUST be `import * as schema from "@/db/schema"` — passing only some tables causes runtime error `TypeError: undefined is not an object (evaluating 'e._.fullSchema')`.

       **Critical:** `nextCookies()` MUST be the last plugin. If placed before other plugins, server actions fail to set cookies.

    2. Create `src/lib/auth-client.ts` — Better Auth browser client:
       ```typescript
       import { createAuthClient } from "better-auth/react";

       export const authClient = createAuthClient({
         baseURL: process.env.NEXT_PUBLIC_APP_URL,
       });
       ```

       This exports `authClient` which provides:
       - `authClient.signIn.email({ email, password })` — returns `{ data, error }`
       - `authClient.signUp.email({ email, password, name })` — returns `{ data, error }`

       This mounts Better Auth's handler on all `/api/auth/*` routes (signin, signup, session, password reset, etc.).

    2. Create `src/proxy.ts` (NOT `middleware.ts` — Next.js 16 breaking change):
       ```typescript
       import { getSessionCookie } from "better-auth/cookies";
       import { type NextRequest, NextResponse } from "next/server";

       export function proxy(request: NextRequest) {
         const sessionCookie = getSessionCookie(request);
         const { pathname } = request.nextUrl;

         // Unauthenticated users trying to access protected routes → redirect to /login with return URL
         if (!sessionCookie && pathname.startsWith("/dashboard")) {
           const redirectUrl = new URL("/login", request.url);
           redirectUrl.searchParams.set("callbackUrl", pathname);
           return NextResponse.redirect(redirectUrl);
         }

         // Authenticated users on auth pages → redirect to /dashboard
         if (sessionCookie && (pathname === "/login" || pathname === "/signup")) {
           return NextResponse.redirect(new URL("/dashboard", request.url));
         }

         // Root route → redirect to /login (no landing page in v1)
         if (pathname === "/") {
           return NextResponse.redirect(new URL("/login", request.url));
         }

         return NextResponse.next();
       }

       export const config = {
         matcher: ["/", "/dashboard/:path*", "/login", "/signup"],
       };
       ```

       **Critical:** Export `function proxy` not `function middleware` — Next.js 16 renamed this.
       **Critical:** File is `src/proxy.ts` not `src/middleware.ts` — deprecated filename emits warnings on every request.
       **Note:** This is an optimistic cookie-presence check only. Protected server components MUST also call `auth.api.getSession()` for real validation.
