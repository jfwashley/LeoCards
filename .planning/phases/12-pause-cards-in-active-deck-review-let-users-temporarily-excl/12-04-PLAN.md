        const [pendingCardIds, setPendingCardIds] = useState<Set<string>>(() => new Set());
        const [, startTransition] = useTransition();

        const togglePause = (card: CardRow) => {
          setPendingCardIds((prev) => new Set(prev).add(card.id));
          startTransition(async () => {
            const action = card.pausedAt ? "unpause" : "pause";
            try {
              const res = await fetch(`/api/cards/${card.id}/${action}`, { method: "POST" });
              if (res.ok) {
                router.refresh(); // Pitfall 2 — revalidatePath alone would not update the open tab
              } else {
                console.error(`Pause toggle failed (${res.status}); state will resync on next refresh`);
              }
            } catch (err) {
              console.error("Pause toggle network error", err);
            } finally {
              setPendingCardIds((prev) => {
                const next = new Set(prev);
                next.delete(card.id);
                return next;
              });
            }
          });
        };

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client `<CardList>` → POST /api/cards/[id]/{pause,unpause} | The same trust boundary already covered by Plan 12-03's threat model — client passes only the card id; server enforces auth + ownership + rate-limit. |
| Server-rendered `dashboard/page.tsx` → `<DeckView>` props | Server-side; no untrusted input crosses (session derives from cookies, validated by Better Auth before any DB read). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-11 | Tampering | Client forging card id in the fetch URL | mitigate | All authority is server-side (Plan 12-03's ownership check). The client component holds no privilege — guessing another user's card id results in 403 with no state change. |
| T-12-12 | Information Disclosure | Logging card ids / pause state to the console | accept | console.error on failure logs the HTTP status only (no card content, no pausedAt timestamp). Acceptable for client-side diagnostic logging; the project already logs similar at this level. |
| T-12-13 | Denial of Service | Click-storm pausing many cards rapidly | mitigate | Server-side rate-limit (30/min/user from Plan 12-03) + per-card disabled-while-pending state on the client prevents a single row from re-firing before its in-flight POST resolves. |
| T-12-14 | (Accessibility, not STRIDE — tracked here for completeness) | Pause button keyboard reachability | mitigate | Reuses the existing `<Button variant="ghost">` shadcn primitive which is keyboard-focusable; aria-label and title set per state; matches the existing Pencil button's a11y precedent (line 148). |
</threat_model>
