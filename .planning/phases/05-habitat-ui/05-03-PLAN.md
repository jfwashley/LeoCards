   On mount (useEffect), cache the server-provided state to localStorage:
   ```typescript
   const CACHE_KEY = "tiocards:habitat-state";
   useEffect(() => {
     try { localStorage.setItem(CACHE_KEY, JSON.stringify(habitatState)); } catch {}
   }, [habitatState]);
   ```

   Add a `retry` function for the error state:
   ```typescript
   async function retry() {
     setError(false);
     try {
       const res = await fetch("/api/habitat");
       if (!res.ok) throw new Error("API error");
       const data: HabitatState = await res.json();
       setState(data);
       localStorage.setItem(CACHE_KEY, JSON.stringify(data));
       setOffline(false);
     } catch {
       // Try cached data
       const cached = localStorage.getItem(CACHE_KEY);
       if (cached) {
         setState(JSON.parse(cached) as HabitatState);
         setOffline(true);
       } else {
         setError(true);
       }
     }
   }
   ```

   Level-up detection: when `state.level > prevLevel`, set `showLevelUp = true`, after 2.5s set it back to false:
   ```typescript
   useEffect(() => {
     if (state.level > prevLevel) {
       setShowLevelUp(true);
       setPrevLevel(state.level);
       const timer = setTimeout(() => setShowLevelUp(false), 2500);
       return () => clearTimeout(timer);
     }
     setPrevLevel(state.level);
   }, [state.level, prevLevel]);
   ```

   Render structure:
   ```
   if (error):
     <div className="w-full aspect-[16/9] max-h-[70vh] flex flex-col items-center justify-center bg-card rounded-lg border">

       <p className="text-sm text-muted-foreground mb-4">We couldn't load your habitat. Check your connection and try again.</p>
       <Button onClick={retry}>Try again</Button>
     </div>

   else:
     <div className="relative w-full">
       <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.5 }}>
         <HabitatCanvas habitatState={state} />
       </motion.div>

       <!-- Level badge overlay (already added in Plan 02) -->

       {offline && (
         <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.3 }}
           className="absolute bottom-3 left-1/2 -translate-x-1/2 bg-card/90 backdrop-blur-sm text-sm text-muted-foreground px-4 py-2 rounded-full border">
           You're offline — showing last known state
         </motion.div>
       )}

       {showLevelUp && (
         <motion.div
           initial={{ opacity: 0, scale: 0.8 }}
           animate={{ opacity: 1, scale: 1 }}
           exit={{ opacity: 0 }}
           transition={{ duration: 0.6, ease: "easeOut" }}
   ```
