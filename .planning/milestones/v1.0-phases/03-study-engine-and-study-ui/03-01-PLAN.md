```typescript
import type { CardId, RecallDirection } from "@/db/schema";

export interface CardForSession {
  id: CardId;
  front: string;
  back: string;
  masteryRound: number; // 0|1|2|3
  cooldownUntil: Date | null;
  createdAt: Date;
  isResurface: boolean;
}

export interface SessionCard extends CardForSession {
  stage: "n2t" | "t2n";
}

export type GradeEntry = {
  cardId: CardId;
  direction: RecallDirection;
  correct: boolean;
};

export type SessionStats = {
  cardsStudied: number;
  correctCount: number;
  newlyLearned: number;
};
```

Implement these functions:

1. `assembleSession(cards: CardForSession[], now: Date): SessionCard[]`
   - Filter due unlearned: cards where masteryRound < 3 AND (cooldownUntil is null OR cooldownUntil <= now)
   - Sort newest-first by createdAt descending (D-05)
   - Compute resurface pool: ~10% of due unlearned count, min 1 if any learned exist
   - Learned = cards where masteryRound === 3
   - Shuffle learned, take resurfaceCount, mark isResurface = true
   - Call interleave() to mix them
   - Assign stage via getCardStage() to produce SessionCard[]

2. `interleave(learning: CardForSession[], resurface: CardForSession[], interval: number): CardForSession[]`
   - Insert one resurface card every `interval` learning cards (interval=3)
   - If resurface runs out, append remaining learning cards
   - If learning runs out, append remaining resurface cards

3. `getCardStage(card: CardForSession): "n2t" | "t2n"`
   - Round 0: return "n2t"
   - Round 1: return "t2n"
   - Round 2 or 3: return Math.random() < 0.5 ? "n2t" : "t2n"

4. `computeCardUpdate(cardId: CardId, currentRound: number, grades: GradeEntry[], now: Date): { newRound: number; cooldownUntil: Date | null; recallCountDelta: number }`
   - Round thresholds: { 0: {n2t:2, t2n:2}, 1: {n2t:1, t2n:1}, 2: {n2t:1, t2n:1} }
   - Cooldowns: { 0: 12*3600*1000, 1: 24*3600*1000 } (no cooldown after round 2->3)
   - Count correct grades per direction. Cap at threshold to prevent multi-round advancement in single session (Pitfall 4).
   - If thresholds met: newRound = currentRound + 1, set cooldown. Else: newRound = currentRound, cooldown = null.
   - recallCountDelta = total correct grades (uncapped — for habitat progression count).

5. `earliestCooldownEnd(cards: CardForSession[], now: Date): Date | null`
   - Filter cards with cooldownUntil > now
   - Return the minimum date, or null if none

Use a simple `shuffleTake` helper (Fisher-Yates on a copy, take first N). Export it for testing.
