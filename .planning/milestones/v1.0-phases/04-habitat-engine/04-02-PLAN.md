    **Implementation:**

    ```typescript
    import { headers } from "next/headers";
    import { auth } from "@/lib/auth";
    import { getHabitatFacts } from "@/lib/habitat-queries";
    import { computeHabitatState } from "@/lib/habitat-engine";
    import type { UserId } from "@/db/schema";

    export async function GET() {
      // 1. Auth check (MUST await headers() — Next.js 16, Pitfall 5)
      const session = await auth.api.getSession({ headers: await headers() });
      if (!session) {
        return Response.json({ error: "Unauthorized" }, { status: 401 });
      }

      // 2. Fetch raw facts from DB
      const facts = await getHabitatFacts(session.user.id as UserId);

      // 3. Compute habitat state (pure function, no side effects)
      const state = computeHabitatState(facts, new Date());

      // 4. Return typed JSON
      return Response.json(state);
    }
    ```

    **Key points:**
    - This is a GET-only route. No POST, no request body parsing, no Zod validation needed (D-15).
    - `await headers()` is REQUIRED in Next.js 16 (Pitfall 5 from research). Copy exact pattern from study/complete/route.ts.
    - No writes to DB — this is a pure read endpoint per compute-on-read architecture.
    - No try/catch around getHabitatFacts — let Next.js error boundary handle unexpected DB errors (consistent with existing routes).
    - Cast `session.user.id as UserId` — branded type pattern from Phase 2.
